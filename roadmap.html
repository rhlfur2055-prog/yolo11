AI Hub → YOLO 포맷 변환
convert_aihub_to_yolo.py Python 
import json, os, glob
from pathlib import Path

def convert_aihub_to_yolo(json_dir, output_dir, img_w=3840, img_h=2160):
    """AI Hub JSON → YOLO txt 변환

    시간복잡도: O(n) -- JSON 파일 수에 비례

    AI Hub bbox: [x, y, width, height] (픽셀)
    YOLO bbox:   [x_center, y_center, w, h] (0~1 정규화)
    """
    os.makedirs(output_dir, exist_ok=True)

    json_files = glob.glob(os.path.join(json_dir, "*.json"))
    print(f"변환할 JSON: {len(json_files)}개")

    for jf in json_files:
        with open(jf, 'r', encoding='utf-8') as f:
            data = json.load(f)

        # 라벨 파일명 = JSON 파일명.txt
        txt_name = Path(jf).stem + ".txt"
        txt_path = os.path.join(output_dir, txt_name)

        lines = []
        annotations = data.get("Learning_Data_Info", {}).get("annotations", [])

        for ann in annotations:
            plates = ann.get("license_plate", [])
            for plate in plates:
                bbox = plate.get("bbox")
                if not bbox: continue

                x, y, w, h = bbox
                # 핵심 변환: 픽셀 → 정규화 중심좌표
                x_center = (x + w / 2) / img_w
                y_center = (y + h / 2) / img_h
                w_norm = w / img_w
                h_norm = h / img_h

                # class 0 = license_plate
                lines.append(f"0 {x_center:.6f} {y_center:.6f} {w_norm:.6f} {h_norm:.6f}")

        with open(txt_path, 'w') as f:
            f.write("\n".join(lines))

    print(f"변환 완료: {output_dir}")

# 사용법
convert_aihub_to_yolo(
    json_dir="./aihub_data/labels_json",
    output_dir="./dataset/labels/train"
)

# yolo11n.pt -> 가장 빠름 (5.4MB), 정확도 낮음
# yolo11s.pt -> 균형형 (18.5MB) <- 추천!

model = YOLO("yolo11s.pt")  # s = small, 균형 좋음

# 2. Fine-tuning 시작!
results = model.train(
    data="dataset/data.yaml",

    # === 학습 설정 ===
    epochs=100,            # 에포크 (50~150 권장)
    imgsz=640,             # 입력 크기 (640 기본)
    batch=16,              # 배치 크기 (GPU 메모리에 맞게)

    # === 최적화 ===
    optimizer="AdamW",
    lr0=0.001,             # 초기 학습률
    lrf=0.01,              # 최종 학습률 비율
    warmup_epochs=3,       # 워밍업

    # === 증강 (Augmentation) ===
    hsv_h=0.015,           # 색조 변화
    hsv_s=0.7,             # 채도 변화
    hsv_v=0.4,             # 명도 변화
    mosaic=1.0,            # 모자이크 증강
    flipud=0.0,            # 상하 반전 OFF (번호판은 뒤집으면 안됨!)
    fliplr=0.0,            # 좌우 반전 OFF (번호판 좌우 반전 X)

    # === 저장 ===
    project="runs/train",
    name="anpr_v1",
    save=True,
    plots=True,
)

# 3. 최고 성능 모델 확인
print(f"Best model: runs/train/anpr_v1/weights/best.pt")
print(f"mAP50: {results.results_dict['metrics/mAP50(B)']:.4f}")
IMPORTANT
"이거 틀리면 99% 이 부분" -- YOLO 학습 5대 실수:
1. flipud/fliplr=1.0 → 번호판 뒤집으면 OCR 망가짐. 반드시 0.0으로!
2. 이미지-라벨 파일명 불일치 → img001.jpg ↔ img001.txt 1:1 매칭 필수
3. bbox 좌표 1 초과 → YOLO 정규화값은 0~1 사이여야 함
4. batch 너무 큼 → CUDA OOM 에러 → batch 줄이기 (16→8→4)
5. data.yaml 경로 오류 → 절대경로 or 실행 디렉토리 기준 상대경로 확인

src/preprocessor.py -- 7종 이미지 전처리 Python 
import cv2
import numpy as np

class PlatePreprocessor:
    """번호판 이미지 전처리 7종 세트

    비유: 사진관에서 증명사진 보정하듯이,
    번호판 이미지도 OCR이 잘 읽을 수 있도록
    여러 가지 방법으로 "보정"하는 것!

    시간복잡도: O(w*h) -- 이미지 픽셀 수에 비례
    """

    @staticmethod
    def deblur(img):
        """1. 샤프닝 (흐릿한 이미지 선명하게)"""
        kernel = np.array([[-1,-1,-1],
                          [-1, 9,-1],
                          [-1,-1,-1]])
        return cv2.filter2D(img, -1, kernel)

    @staticmethod
    def gamma_correct(img, gamma=1.5):
        """2. 감마 보정 (어두운 이미지 밝게)"""
        table = np.array([
            ((i / 255.0) ** (1.0 / gamma)) * 255
            for i in range(256)
        ]).astype("uint8")
        return cv2.LUT(img, table)

    @staticmethod
    def contrast_stretch(img):
        """3. 히스토그램 스트레칭 (대비 극대화)"""
        lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
        l, a, b = cv2.split(lab)
        l = cv2.normalize(l, None, 0, 255, cv2.NORM_MINMAX)
        return cv2.cvtColor(cv2.merge([l, a, b]), cv2.COLOR_LAB2BGR)

    @staticmethod
    def clahe(img):
        """4. CLAHE (지역 적응형 히스토그램 평활화)"""
        lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
        l, a, b = cv2.split(lab)
        clahe = cv2.createCLAHE(clipLimit=3.0, tileGridSize=(8, 8))
        l = clahe.apply(l)
        return cv2.cvtColor(cv2.merge([l, a, b]), cv2.COLOR_LAB2BGR)

    @staticmethod
    def bilateral_filter(img):
        """5. 양방향 필터 (노이즈 제거 + 엣지 보존)"""
        return cv2.bilateralFilter(img, 9, 75, 75)

    @staticmethod
    def morphology(img):
        """6. 모폴로지 연산 (문자 윤곽 강화)"""
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (3, 3))
        return cv2.morphologyEx(gray, cv2.MORPH_CLOSE, kernel)

    @staticmethod
    def adaptive_threshold(img):
        """7. 적응형 이진화 (배경/글자 완전 분리)"""
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        return cv2.adaptiveThreshold(
            gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
            cv2.THRESH_BINARY, 11, 2
        )

    def get_all_versions(self, img):
        """7종 전처리 결과를 모두 반환"""
        return {
            "original": img,
            "deblur": self.deblur(img),
            "gamma": self.gamma_correct(img),
            "stretch": self.contrast_stretch(img),
            "clahe": self.clahe(img),
            "bilateral": self.bilateral_filter(img),
            "morph": self.morphology(img),
            "threshold": self.adaptive_threshold(img),
        }
src/ocr_pipeline.py -- 멀티 OCR 엔진 Python 
from paddleocr import PaddleOCR
import easyocr
import pytesseract
import re

class MultiOCR:
    """3종 OCR 엔진 체이닝

    비유: 3명의 전문가에게 같은 사진을 보여주고
    가장 자신있는 답을 채택하는 것!

    PaddleOCR: 한글에 강함 (1순위)
    EasyOCR:   영문+한글 (2순위)
    Tesseract: 범용 폴백 (3순위)
    """

    def __init__(self):
        self.paddle = PaddleOCR(
            lang="korean",
            use_angle_cls=True,
            show_log=False
        )
        self.easy = easyocr.Reader(['ko', 'en'], gpu=True)

    def _clean_plate_text(self, text):
        """한국 번호판 포맷 정리
        예: '12가 3456' → '12가3456'
        """
        text = re.sub(r'[\s\-\.]', '', text)
        text = text.strip()
        return text

    def _validate_korean_plate(self, text):
        """한국 번호판 패턴 검증

        신규: 123가4567 (3자리+한글+4자리)
        기존: 12가3456  (2자리+한글+4자리)
        """
        patterns = [
            r'^\d{2,3}[가-힣]\d{4}$',   # 기본
            r'^[가-힣]{2}\d{2}[가-힣]\d{4}$',  # 지역명 포함
        ]
        return any(re.match(p, text) for p in patterns)

    def recognize(self, images_dict):
        """전처리된 이미지들로 OCR 실행

        Args:
            images_dict: preprocessor.get_all_versions() 결과
        Returns:
            (text, confidence, engine) 최적 결과
        """
        results = []

        for name, img in images_dict.items():
            # PaddleOCR (1순위)
            try:
                paddle_res = self.paddle.ocr(img, cls=True)
                if paddle_res and paddle_res[0]:
                    for line in paddle_res[0]:
                        text = self._clean_plate_text(line[1][0])
                        conf = line[1][1]
                        results.append((text, conf, "paddle", name))
            except: pass

            # EasyOCR (2순위)
            try:
                easy_res = self.easy.readtext(img)
                for (_, text, conf) in easy_res:
                    text = self._clean_plate_text(text)
                    results.append((text, conf, "easyocr", name))
            except: pass

        # 검증된 결과 우선, 그 다음 confidence 순
        valid = [r for r in results if self._validate_korean_plate(r[0])]

        if valid:
            best = max(valid, key=lambda x: x[1])
        elif results:
            best = max(results, key=lambda x: x[1])
        else:
            best = ("", 0.0, "none", "none")

        return {
            "text": best[0],
            "confidence": best[1],
            "engine": best[2],
            "preprocess": best[3],
            "is_valid": self._validate_korean_plate(best[0])
        }

src/plate_recognizer.py -- 통합 엔진 Python 
from ultralytics import YOLO
from src.preprocessor import PlatePreprocessor
from src.ocr_pipeline import MultiOCR
import cv2, json, os
from datetime import datetime

class PlateRecognizer:
    """완전한 번호판 인식 파이프라인

    비유: 택배 분류 센터와 같음
    1. YOLO = 컨베이어벨트에서 택배 찾기
    2. 전처리 = 라벨 닦아서 깨끗하게
    3. OCR = 바코드 스캔해서 텍스트 읽기
    4. 검증 = 올바른 주소인지 확인
    """

    def __init__(self, model_path="models/best.pt"):
        self.detector = YOLO(model_path)
        self.preprocessor = PlatePreprocessor()
        self.ocr = MultiOCR()
        self.results = []

    def process_frame(self, frame, conf_threshold=0.5):
        """단일 프레임 처리 -- 시간복잡도: O(n*w*h)
        n = 탐지된 번호판 수, w*h = 이미지 크기
        """
        detections = self.detector(frame, conf=conf_threshold)[0]
        frame_results = []

        for box in detections.boxes:
            x1, y1, x2, y2 = map(int, box.xyxy[0])
            conf = float(box.conf[0])

            # 번호판 영역 crop
            plate_img = frame[y1:y2, x1:x2]
            if plate_img.size == 0: continue

            # 7종 전처리 적용
            versions = self.preprocessor.get_all_versions(plate_img)

            # 멀티 OCR로 텍스트 인식
            ocr_result = self.ocr.recognize(versions)

            frame_results.append({
                "bbox": [x1, y1, x2, y2],
                "det_conf": conf,
                "plate_text": ocr_result["text"],
                "ocr_conf": ocr_result["confidence"],
                "engine": ocr_result["engine"],
                "is_valid": ocr_result["is_valid"],
                "timestamp": datetime.now().isoformat()
            })

        return frame_results

    def process_video(self, video_path, save_dir="results"):
        """비디오 전체 처리"""
        os.makedirs(save_dir, exist_ok=True)
        cap = cv2.VideoCapture(video_path)
        fps = cap.get(cv2.CAP_PROP_FPS)
        frame_count = 0
        all_results = []

        while cap.isOpened():
            ret, frame = cap.read()
            if not ret: break

            # 매 5프레임마다 처리 (속도 최적화)
            if frame_count % 5 == 0:
                results = self.process_frame(frame)
                for r in results:
                    r["frame"] = frame_count
                    r["time"] = frame_count / fps
                    all_results.append(r)
                    print(f"[{r['time']:.1f}s] {r['plate_text']} "
                          f"(conf: {r['ocr_conf']:.2f}, {r['engine']})")

            frame_count += 1

        cap.release()

        # 결과 저장
        with open(f"{save_dir}/results.json", "w") as f:
            json.dump(all_results, f, ensure_ascii=False, indent=2)

        print(f"\n총 {len(all_results)}개 번호판 인식 완료")
        return all_results
